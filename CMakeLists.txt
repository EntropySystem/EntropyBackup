cmake_minimum_required(VERSION 3.24)
project(EntropyBackup VERSION 1.5.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(EntropyBackup WIN32 src/main.cpp)
target_compile_definitions(EntropyBackup PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(EntropyBackup PRIVATE d2d1 dwrite windowscodecs shlwapi comctl32 shell32 ole32 uuid dwmapi uxtheme)
if(MSVC)
  target_compile_options(EntropyBackup PRIVATE /W4 /permissive- /EHsc /utf-8)
  set_property(TARGET EntropyBackup PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
